    StartCoroutine(_apiClient.FetchUnitySession(sid, data => Apply(data, onLoaded)));

    private IEnumerator LoadServerSession(string sid, Action<SessionData> onLoaded)

## 버그 A — 오브제 비동기 타이밍 (GameFlowManager.cs)

[세션 11] MissingComponentException 수정
────────────────────────────────────────────────────────────────

## 에러
  MissingComponentException: There is no 'Rigidbody' attached to the
  "PlazaObj_plaza_001" game object
  LegoTwin.Plaza.PlazaManager.SetupObjectPhysics() 에서 발생

## 원인
  Unity Object에 C# ?? 연산자 사용
  → ??는 ReferenceEquals 기반 → Unity null 오브젝트를 null로 인식 못 함
  → GetComponent<Rigidbody>()가 Unity null 반환해도 ??가 AddComponent를 건너뜀
  → null Rigidbody에 .useGravity 접근 → 예외 발생

## 해결
  ?? 연산자 → if (rb == null) 패턴으로 교체 (Unity == 오버로드 활용)

  변경 전:
    var rb = go.GetComponent<Rigidbody>() ?? go.AddComponent<Rigidbody>();
    var col = mf.gameObject.GetComponent<MeshCollider>() ?? mf.gameObject.AddComponent<MeshCollider>();

  변경 후:
    var rb = go.GetComponent<Rigidbody>();
    if (rb == null) rb = go.AddComponent<Rigidbody>();
    var col = mf.gameObject.GetComponent<MeshCollider>();
    if (col == null) col = mf.gameObject.AddComponent<MeshCollider>();
